import sqlite3, statistics
FX = {"USD": 1.0, "EUR": 1.09}

def scout(sku):
    con = sqlite3.connect("local_db.sqlite")
    rows = con.execute("SELECT competitor, price, currency, status FROM mock_prices WHERE sku=?", (sku,)).fetchall()
    con.close()
    return [{"n": n, "usd": round(p * FX[c], 2), "s": s} for n, p, c, s in rows]

def strategist(our_price, comps):
    med = statistics.median(c["usd"] for c in comps)
    bad = [c for c in comps if c["usd"] < 0.5 * med]
    ok = [c for c in comps if c not in bad]
    oos = [c for c in ok if c["s"] == "Out of Stock"]
    norm = [c for c in ok if c["s"] == "In Stock"]
    low = min(norm, key=lambda c: c["usd"], default=None)
    if oos and (not low or low["usd"] >= our_price):
        return {"action": "RAISE_PRICE", "target": round(our_price * 1.08, 2)}
    if low and low["usd"] < our_price:
        return {"action": "MATCH", "target": round(low["usd"] - 0.01, 2)}
    return {"action": "HOLD", "target": our_price}

def guardrail(cogs, day_start, target, margin=0.15, cap=0.10):
    floor = cogs / (1 - margin)
    lo, hi = day_start * (1 - cap), day_start * (1 + cap)
    v = []
    if target < floor: v.append("margin floor")
    if not lo <= target <= hi: v.append("daily cap")
    return {"approved": not v, "violations": v}

def update_price(sku, price):
    con = sqlite3.connect("local_db.sqlite")
    con.execute("UPDATE store_inventory SET price=? WHERE sku=?", (price, sku))
    con.commit(); con.close()
